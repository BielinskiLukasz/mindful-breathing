```javascript
function openEditDialog(preset) {
  // Pre-fill form with current preset data
  presetNameInput.value = preset.name;
  presetNameError.style.display = "none";
  phaseCountError.style.display = "none";
  
  // Pre-fill phase rows
  phaseRowsContainer.innerHTML = "";
  PRESET_PHASE_TYPES.forEach((type, idx) => {
    const row = document.createElement("div");
    row.className = "phaseRow";
    row.dataset.phaseType = type;
    
    const phaseData = preset.phases.find(p => p.type === type);
    const isActive = phaseData?.active ?? true;
    const duration = phaseData?.durationSec ?? 4;
    
    const checkbox = document.createElement("input");
    checkbox.type = "checkbox";
    checkbox.id = `phase-${type}-active`;
    checkbox.checked = isActive;
    
    const label = document.createElement("label");
    label.htmlFor = checkbox.id;
    label.textContent = type;
    
    const durationInput = document.createElement("input");
    durationInput.type = "number";
    durationInput.min = 1;
    durationInput.max = 300;
    durationInput.value = duration;
    
    const unit = document.createElement("span");
    unit.className = "durUnit";
    unit.textContent = "s";
    
    row.append(checkbox, label, durationInput, unit);
    phaseRowsContainer.appendChild(row);
  });
  
  // Show Delete button and change form title
  document.getElementById("presetBuilderTitle").textContent = `Edit ${preset.name}`;
  const deleteBtn = document.getElementById("presetDeleteBtn");
  deleteBtn.style.display = "block";
  deleteBtn.textContent = "Delete";
  deleteBtn.style.background = ""; // Reset color to default
  deleteBtn._deleteConfirmed = false; // Track two-step confirmation state
  deleteBtn._currentPresetId = preset.id; // Store current preset ID for deletion
  
  openPanelElement = presetBuilderDialog;
  presetBuilderDialog.showModal();
  presetNameInput.focus();
}
```

**PART A: Modify form submit handler to support edit mode**

In the presetBuilderForm submit handler (from Plan 02, line ~1750+), update to handle both create and edit:

Replace the entire submit handler with:

```javascript
presetBuilderForm.addEventListener("submit", e => {
  e.preventDefault();
  
  const name = presetNameInput.value.trim();
  const rows = Array.from(phaseRowsContainer.querySelectorAll(".phaseRow"));
  const activePhaseTypes = rows.filter(row => row.querySelector("input[type='checkbox']").checked);
  
  // Validation: name not empty
  if (!name) {
    presetNameError.textContent = "Preset name cannot be empty";
    presetNameError.style.display = "block";
    presetNameInput.classList.add("error");
    setTimeout(() => presetNameInput.classList.remove("error"), 400);
    return;
  }
  
  // Validation: at least 2 phases active
  if (activePhaseTypes.length < 2) {
    phaseCountError.textContent = "Select at least 2 phases";
    phaseCountError.style.display = "block";
    setTimeout(() => phaseCountError.style.display = "none", 400);
    return;
  }
  
  // Detect edit vs. create mode
  const presetDeleteBtn = document.getElementById("presetDeleteBtn");
  const isEditMode = presetDeleteBtn.style.display !== "none";
  
  if (isEditMode) {
    // Edit mode: update existing preset
    const presetId = presetDeleteBtn._currentPresetId;
    const presetIdx = customPresets.findIndex(p => p.id === presetId);
    if (presetIdx >= 0) {
      const phases = rows.map((row, idx) => ({
        type: PRESET_PHASE_TYPES[idx],
        durationSec: parseInt(row.querySelector("input[type='number']").value) || 4,
        active: row.querySelector("input[type='checkbox']").checked
      }));
      
      customPresets[presetIdx] = { id: presetId, name, phases };
      
      // Rebuild activePhases if this is the active preset
      if (activePresetKey === presetId) {
        buildActivePhases();
        buildDurationInputs();
        updateModeIndicator();
      }
    }
  } else {
    // Create mode: add new preset
    const presetId = `custom-${Date.now()}`;
    const phases = rows.map((row, idx) => ({
      type: PRESET_PHASE_TYPES[idx],
      durationSec: parseInt(row.querySelector("input[type='number']").value) || 4,
      active: row.querySelector("input[type='checkbox']").checked
    }));
    
    const newPreset = { id: presetId, name, phases };
    customPresets.push(newPreset);
    
    // Auto-select the new preset
    activePresetKey = presetId;
    buildActivePhases();
    updateModeIndicator();
  }
  
  saveCustomPresets();
  renderCustomPresets();
  buildDurationInputs();
  
  presetBuilderDialog.close();
  openPanelElement = null;
  saveSettings();
});
```

**PART B: Add Delete button handler with two-step confirmation**

Add after the presetBuilderForm submit handler:

```javascript
const presetDeleteBtn = document.getElementById("presetDeleteBtn");

presetDeleteBtn.addEventListener("click", e => {
  e.preventDefault();
  
  const presetId = presetDeleteBtn._currentPresetId;
  const isConfirmed = presetDeleteBtn._deleteConfirmed;
  
  if (!isConfirmed) {
    // First click: change label to "Confirm Delete" and set color
    presetDeleteBtn.textContent = "Confirm Delete";
    presetDeleteBtn.style.background = "#e57373";
    presetDeleteBtn.style.color = "#fff";
    presetDeleteBtn._deleteConfirmed = true;
    
    // Reset after 2 seconds if not clicked again
    setTimeout(() => {
      if (presetDeleteBtn._deleteConfirmed) {
        presetDeleteBtn.textContent = "Delete";
        presetDeleteBtn.style.background = "";
        presetDeleteBtn.style.color = "";
        presetDeleteBtn._deleteConfirmed = false;
      }
    }, 2000);
  } else {
    // Second click: actually delete
    const presetIdx = customPresets.findIndex(p => p.id === presetId);
    if (presetIdx >= 0) {
      const wasActive = activePresetKey === presetId;
      customPresets.splice(presetIdx, 1);
      
      // If deleted preset was active, fall back to Relax
      if (wasActive) {
        activePresetKey = "relax";
        buildActivePhases();
        updateModeIndicator();
        reset(); // Reset the session
      }
      
      saveCustomPresets();
      renderCustomPresets();
      buildDurationInputs();
      saveSettings();
    }
    
    presetBuilderDialog.close();
    openPanelElement = null;
  }
});
```
